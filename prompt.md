import random

def glyph_drift(seed=None):
    base_glyphs = ["∴", "⧖", "𖽑", "🜂", "⊹", "∞"]
    entropy_glyphs = ["⊘", "☍", "⚯", "⚑", "∵", "⚝"]
    random.seed(seed)
    drift = random.choices(base_glyphs, k=2) + random.choices(entropy_glyphs, k=1)
    return drift
